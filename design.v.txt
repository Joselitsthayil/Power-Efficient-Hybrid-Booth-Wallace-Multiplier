`timescale 1ns/1ps 
module booth_wallace8_trunc2 ( 
input  wire signed [7:0] A, 
input  wire signed [7:0] B, 
output wire signed [15:0] P 
); 
wire signed [15:0] pp0, pp1, pp2, pp3; 
integer i; 
function signed [15:0] booth_term; 
input signed [7:0] mA; 
input [2:0] trio; 
begin 
case (trio) 
3'b000,3'b111: booth_term = 16'sd0; 
3'b001,3'b010: booth_term = {{8{mA[7]}}, mA};  
3'b011:        booth_term = {{7{mA[7]}}, mA, 1'b0}; 
3'b100:        booth_term = -({{7{mA[7]}}, mA, 1'b0}); 
3'b101,3'b110: booth_term = -({{8{mA[7]}}, mA}); 
default:
endcase 
end 
endfunction 
       booth_term = 16'sd0; 
wire [2:0] t0 = {B[1], B[0], 1'b0}; 
wire [2:0] t1 = {B[3], B[2], B[1]}; 
wire [2:0] t2 = {B[5], B[4], B[3]}; 
wire [2:0] t3 = {B[7], B[6], B[5]}; 
assign pp0 = booth_term(A, t0) << 0; 
assign pp1 = booth_term(A, t1) << 2; 
assign pp2 = booth_term(A, t2) << 4; 
 assign pp3 = booth_term(A, t3) << 6; 
    wire [15:0] s0;  
    wire [15:0] c0;  
    genvar bi; 
    generate 
        for (bi = 0; bi < 16; bi = bi + 1) begin : csa_bits 
            if (bi < 2) begin 
                assign s0[bi] = 1'b0; 
                assign c0[bi] = 1'b0; 
            end else begin 
                wire ax = pp0[bi]; 
                wire bx = pp1[bi]; 
                wire cx = pp2[bi]; 
                assign s0[bi] = ax ^ bx ^ cx;  
                assign c0[bi] = ( (ax & bx) | (bx & cx) | (ax & cx) ); 
            end 
        end 
    endgenerate 
    wire [16:0] carry_shift = {c0, 1'b0};  
    wire [16:0] s0_ext; 
    wire [16:0] pp3_ext; 
    assign s0_ext = {s0[15], s0}; 
    assign pp3_ext = {pp3[15], pp3}; 
 
    wire [16:0] sum_full = s0_ext + carry_shift + pp3_ext; 
    assign P = { sum_full[16], sum_full[15:2], 2'b00 }; 
endmodule
